    <div id="result"></div>

  <script>
   
    const decryptMap = {
     "B" : "А",
   "." : "Б",
   "D" : "В",
   "," : "Г",
   "F" : "Д",
   "!" : "Е",
   "H" : "Ж",
   "?" : "З",
   "J" : "И",
   "*" : "К",
   "L" : "Л",
   ":" : "М",
   "N" : "Н",
   "@" : "О",
   "P" : "П",
   "%" : "Р",
   "R" : "С",
   "-" : "Т",
   "T" : "У",
   "_" : "Ф",
   "V" : "Х",
   "=" : "Ц",
   "W" : "Ч",
   "+" : "Ш",
   "U" : "Щ",
   "(" : "Ы",
   "Z" : "Ь",
   ")" : "Э",
   "|" : "Ю",
   "/" : "Я"
    };

    function decrypt() {
      const input = document.getElementById('input').value;
      let output = '';

      for (const ch of input) {
        if (decryptMap.hasOwnProperty(ch)) {
          output += decryptMap[ch];
        } else {
          output += ch; // оставляем как есть (пробелы, цифры и т.д.)
        }
      }

      document.getElementById('result').textContent = output;
    }
  </script>
